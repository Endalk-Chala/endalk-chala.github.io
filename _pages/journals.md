function getEl(id) { return document.getElementById(id); }

function renderJournals() {
  var search = getEl('search-input').value.toLowerCase();
  var sf = getEl('filter-subfield').value;
  var oa = getEl('filter-oa').value;
  var idx = getEl('filter-indexed').value;
  var sort = getEl('sort-select').value;

  var filtered = journals.filter(function(j) {
    if (search && !(j.name.toLowerCase().indexOf(search) >= 0 || 
        j.publisher.toLowerCase().indexOf(search) >= 0 || 
        j.description.toLowerCase().indexOf(search) >= 0 ||
        j.subfield.toLowerCase().indexOf(search) >= 0)) return false;
    if (sf && j.subfield !== sf) return false;
    if (oa && j.open_access !== oa) return false;
    if (idx && j.indexed.indexOf(idx) < 0) return false;
    return true;
  });

  filtered.sort(function(a, b) {
    if (sort === 'name') return a.name.localeCompare(b.name);
    if (sort === 'name-desc') return b.name.localeCompare(a.name);
    if (sort === 'subfield') return a.subfield.localeCompare(b.subfield) || a.name.localeCompare(b.name);
    var aIF = a.impact_factor === 'N/A' ? -1 : parseFloat(a.impact_factor);
    var bIF = b.impact_factor === 'N/A' ? -1 : parseFloat(b.impact_factor);
    if (sort === 'impact-desc') return bIF - aIF;
    if (sort === 'impact-asc') return aIF - bIF;
    return 0;
  });

  getEl('result-count').textContent = 'Showing ' + filtered.length + ' of ' + journals.length + ' journals';

  var html = '';
  for (var i = 0; i < filtered.length; i++) {
    var j = filtered[i];
    var ifBadge = j.impact_factor !== 'N/A' 
      ? '<span style="background:#2B4C7E;color:#fff;padding:2px 8px;border-radius:12px;font-size:12px;font-weight:600;">IF: ' + j.impact_factor + '</span>' 
      : '';
    var oaBadge = '';
    if (j.open_access === 'Full OA') oaBadge = '<span style="background:#28a745;color:#fff;padding:2px 8px;border-radius:12px;font-size:12px;">Open Access</span>';
    else if (j.open_access === 'Hybrid') oaBadge = '<span style="background:#ffc107;color:#333;padding:2px 8px;border-radius:12px;font-size:12px;">Hybrid OA</span>';
    else oaBadge = '<span style="background:#dc3545;color:#fff;padding:2px 8px;border-radius:12px;font-size:12px;">Subscription</span>';

    html += '<div style="border:1px solid #e0e0e0;border-radius:8px;padding:16px 20px;margin-bottom:12px;background:#fff;transition:box-shadow 0.2s;" onmouseover="this.style.boxShadow=\'0 2px 8px rgba(0,0,0,0.1)\'" onmouseout="this.style.boxShadow=\'none\'">';
    html += '<div style="display:flex;justify-content:space-between;align-items:flex-start;flex-wrap:wrap;gap:8px;">';
    html += '<h3 style="margin:0 0 4px 0;font-size:17px;"><a href="' + j.url + '" target="_blank" style="color:#2B4C7E;text-decoration:none;">' + j.name + '</a></h3>';
    html += '<div style="display:flex;gap:6px;flex-shrink:0;">' + ifBadge + oaBadge + '</div>';
    html += '</div>';
    html += '<div style="font-size:13px;color:#666;margin-bottom:6px;">' + j.publisher + ' &middot; ' + j.subfield + ' &middot; ' + j.frequency + ' &middot; Indexed: ' + j.indexed + '</div>';
    html += '<p style="font-size:14px;line-height:1.5;color:#333;margin:8px 0;">' + j.description + '</p>';
    html += '<a href="' + j.submission_url + '" target="_blank" style="font-size:13px;color:#2B4C7E;">Submission Guidelines &rarr;</a>';
    html += '</div>';
  }

  if (filtered.length === 0) {
    html = '<div style="text-align:center;padding:40px;color:#888;font-size:16px;">No journals match your search criteria. Try adjusting your filters.</div>';
  }
  getEl('journal-list').innerHTML = html;
}

function resetFilters() {
  getEl('search-input').value = '';
  getEl('filter-subfield').value = '';
  getEl('filter-oa').value = '';
  getEl('filter-indexed').value = '';
  getEl('sort-select').value = 'name';
  renderJournals();
}

getEl('search-input').addEventListener('input', renderJournals);
getEl('filter-subfield').addEventListener('change', renderJournals);
getEl('filter-oa').addEventListener('change', renderJournals);
getEl('filter-indexed').addEventListener('change', renderJournals);
getEl('sort-select').addEventListener('change', renderJournals);

renderJournals();
